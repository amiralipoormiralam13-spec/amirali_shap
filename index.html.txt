<script>
const products=[
 {id:1,name:"تیشرت اسپرت",price:450000,category:"لباس",pic:"👕",desc:"تیشرت مناسب استفاده روزمره.",sizes:["S","M","L","XL","XXL"]},

 {id:4,name:"کتونی سفید",price:1450000,category:"کفش",pic:"👟",desc:"مدل ساده و جذاب.",sizes:["۳۸","۳۹","۴۰","۴۱","۴۲","۴۳","۴۴"]},
 {id:5,name:"ساعت مچی",price:900000,category:"اکسسوری",pic:"⌚",desc:"ساعت با طراحی کلاسیک.",sizes:[]},
 {id:6,name:"عینک آفتابی",price:350000,category:"اکسسوری",pic:"🕶️",desc:"عینک با طراحی مدرن.",sizes:[]},
 {id:7,name:"کلاه اسپرت",price:250000,category:"اکسسوری",pic:"🧢",desc:"کلاه مناسب استایل روزمره.",sizes:[]},
 {id:8,name:"شلوار اسپرت",price:700000,category:"لباس",pic:"👖",desc:"شلوار راحت و اسپرت.",sizes:["S","M","L","XL","XXL"]}
];

let category="همه";
let selectedSize="";
let activeProduct=null;
let cart=JSON.parse(localStorage.getItem("amirCart")||"[]");
let favorites=JSON.parse(localStorage.getItem("amirFav")||"[]");

function money(n){return n.toLocaleString("fa-IR")+" تومان"}
function save(){
 localStorage.setItem("amirCart",JSON.stringify(cart));
 localStorage.setItem("amirFav",JSON.stringify(favorites));
}
function notify(message){
 let t=document.getElementById("toast");
 t.textContent=message;t.style.display="block";
 setTimeout(()=>t.style.display="none",1800);
}
function goProducts(){
 document.getElementById("products").scrollIntoView({behavior:"smooth"});
}

function renderProducts(){
 let q=document.getElementById("search").value.trim().toLowerCase();
 let list=products.filter(p=>
  (category==="همه"||p.category===category)&&p.name.toLowerCase().includes(q)
 );
 let sort=document.getElementById("sort").value;
 if(sort==="low")list.sort((a,b)=>a.price-b.price);
 if(sort==="high")list.sort((a,b)=>b.price-a.price);

 let box=document.getElementById("productList");
 if(!list.length){box.innerHTML='<p class="empty">محصولی پیدا نشد!</p>';return}

 box.innerHTML=list.map(p=>`
  <div class="product" onclick="details(${p.id})">
   <div class="pic">${p.pic}</div>
   <h3>${p.name}</h3>
   <p class="price">${money(p.price)}</p>
   <button class="small" onclick="event.stopPropagation();toggleFav(${p.id})">
    ${favorites.includes(p.id)?"❤️":"🤍"}
   </button>
   <button class="primary" onclick="event.stopPropagation();details(${p.id})">
    مشاهده جزئیات
   </button>
  </div>
 `).join("");
}

function setCategory(c,btn){
 category=c;
 document.querySelectorAll(".cat").forEach(x=>x.classList.remove("active"));
 btn.classList.add("active");
 renderProducts();
}

function details(id){
 let p=products.find(x=>x.id===id);
 activeProduct=id;
 selectedSize="";
 let reviews=JSON.parse(localStorage.getItem("reviews_"+id)||"[]");
 let avg=reviews.length?(reviews.reduce((a,r)=>a+r.rating,0)/reviews.length).toFixed(1):"بدون امتیاز";

 document.getElementById("modalContent").innerHTML=`
  <div class="detail-pic">${p.pic}</div>
  <h2>${p.name}</h2>
  <p>${p.desc}</p>
  <p class="price">${money(p.price)}</p>

  <div class="detail-box">
   <h3>📏 سایزهای موجود</h3>
   ${p.sizes.length?`
    <div class="size-buttons">
     ${p.sizes.map(s=>`
      <button class="size-btn" onclick="chooseSize('${s}',this)">${s}</button>
     `).join("")}
    </div>
    <p id="chosenSize">هنوز سایزی انتخاب نکردی.</p>
   `:'<p>این محصول سایزبندی ندارد.</p>'}
  </div>

  <button class="primary" style="width:100%;margin-top:12px"
   onclick="addSelected(${p.id})">افزودن به سبد خرید 🛒</button>

  <div class="detail-box">
   <h3>⭐ امتیاز مشتریان</h3>
   <p>${avg} از ۵ | ${reviews.length} نظر</p>
  </div>

  <div class="detail-box">
   <h3>💬 نظر خودت رو بنویس</h3>
   <input id="reviewName" maxlength="40" placeholder="نام شما">
   <select id="reviewRating">
    <option value="5">⭐⭐⭐⭐⭐ عالی</option>
    <option value="4">⭐⭐⭐⭐ خوب</option>
    <option value="3">⭐⭐⭐ معمولی</option>
    <option value="2">⭐⭐ ضعیف</option>
    <option value="1">⭐ خیلی ضعیف</option>
   </select>
   <textarea id="reviewText" maxlength="500" placeholder="نظرت درباره محصول..."></textarea>
   <button class="primary" onclick="submitReview(${p.id})">ثبت نظر</button>
  </div>

  <div class="detail-box">
   <h3>🗣️ نظرات کاربران</h3>
   <div id="reviewList">
    ${reviews.length?reviews.map(r=>`
     <div class="review">
      <b>${escapeText(r.name)}</b>
      <p>${"⭐".repeat(r.rating)}</p>
      <p>${escapeText(r.text)}</p>
     </div>
    `).join(""):'<p>هنوز نظری ثبت نشده است.</p>'}
   </div>
  </div>
 `;
 document.getElementById("modal").style.display="flex";
}

function chooseSize(size,btn){
 selectedSize=size;
 document.querySelectorAll(".size-btn").forEach(b=>b.classList.remove("selected"));
 btn.classList.add("selected");
 document.getElementById("chosenSize").textContent="سایز انتخاب‌شده: "+size;
}

function addSelected(id){
 let p=products.find(x=>x.id===id);
 if(p.sizes.length&&!selectedSize){
  alert("اول یکی از سایزها رو انتخاب کن!");
  return;
 }
 addCart(id,selectedSize);
 closeModal();
}

function addCart(id,size=""){
 let item=cart.find(x=>x.id===id&&x.size===size);
 if(item)item.qty++;
 else cart.push({id,qty:1,size});
 save();renderCart();notify("به سبد خرید اضافه شد!");
}

function renderCart(){
 let box=document.getElementById("cartList");
 if(!cart.length){
  box.innerHTML='<p class="empty">سبد خرید خالی است.</p>';
  document.getElementById("total").textContent="مجموع: ۰ تومان";
  return;
 }
 let total=0;
 box.innerHTML=cart.map(item=>{
  let p=products.find(x=>x.id===item.id);
  if(!p)return "";
  total+=p.price*item.qty;
  return `<div class="cart-item">
   <div>${p.pic} ${p.name}<br>${item.size?"سایز: "+item.size+"<br>":""}${money(p.price)}</div>
   <div>
    <button onclick="changeQty(${p.id},'${item.size}',-1)">−</button>
    ${item.qty}
    <button onclick="changeQty(${p.id},'${item.size}',1)">+</button>
    <button onclick="removeCart(${p.id},'${item.size}')">حذف</button>
   </div>
  </div>`;
 }).join("");
 document.getElementById("total").textContent="مجموع: "+money(total);
}

function changeQty(id,size,amount){
 let item=cart.find(x=>x.id===id&&x.size===size);
 if(!item)return;
 item.qty+=amount;
 if(item.qty<=0)removeCart(id,size);
 else{save();renderCart()}
}
function removeCart(id,size){
 cart=cart.filter(x=>!(x.id===id&&x.size===size));
 save();renderCart();
}

function toggleFav(id){
 if(favorites.includes(id))favorites=favorites.filter(x=>x!==id);
 else favorites.push(id);
 save();renderProducts();renderFavorites();
}

function renderFavorites(){
 let list=products.filter(p=>favorites.includes(p.id));
 let box=document.getElementById("favoriteList");
 if(!list.length){box.innerHTML='<p class="empty">هنوز محصولی ذخیره نکردی ❤️</p>';return}
 box.innerHTML=list.map(p=>`
  <div class="product" onclick="details(${p.id})">
   <div class="pic">${p.pic}</div>
   <h3>${p.name}</h3>
   <p class="price">${money(p.price)}</p>
   <button class="small" onclick="event.stopPropagation();toggleFav(${p.id})">حذف ❤️</button>
  </div>
 `).join("");
}

function submitReview(id){
 let name=document.getElementById("reviewName").value.trim();
 let text=document.getElementById("reviewText").value.trim();
 let rating=Number(document.getElementById("reviewRating").value);
 if(!name||!text){alert("نام و متن نظر رو وارد کن!");return}
 let reviews=JSON.parse(localStorage.getItem("reviews_"+id)||"[]");
 reviews.unshift({name,text,rating});
 localStorage.setItem("reviews_"+id,JSON.stringify(reviews));
 details(id);
 notify("نظرت ثبت شد ❤️");
}

function escapeText(s){
 return String(s).replace(/[&<>"']/g,c=>({
 "&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#39;"
 }[c]));
}

function closeModal(){document.getElementById("modal").style.display="none"}
document.getElementById("modal").addEventListener("click",e=>{
 if(e.target.id==="modal")closeModal();
});
function showPhone(){document.getElementById("phone").style.display="block"}

function checkout(){
 if(!cart.length){notify("سبد خرید خالی است!");return}
 alert("برای هماهنگی سفارش با شماره ۰۹۹۳۴۵۶۱۲۰۵ تماس بگیر. پرداخت آنلاین هنوز فعال نیست.");
}
